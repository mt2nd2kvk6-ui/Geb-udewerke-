# Gebäudewerke, Serverregeln für Apache (IONOS)
# Stand 28.10.2026
AddDefaultCharset utf-8
AddType font/woff2 .woff2
AddType application/manifest+json .webmanifest
AddCharset utf-8 .txt .xml .webmanifest

DirectoryIndex index.html index.php
ErrorDocument 404 /404.html

<IfModule mod_rewrite.c>
RewriteEngine On

# 1. Immer https und immer gebaeudewerke.de
#    Leitet http, www und jede weitere Domain auf diesem Webspace in einem Schritt um.
RewriteCond %{HTTPS} off [OR]
RewriteCond %{HTTP_HOST} !^gebaeudewerke\.de$ [NC]
RewriteRule ^ https://gebaeudewerke.de%{REQUEST_URI} [L,R=301]

# 2. Doppelte Adressen der Startseiten vermeiden
RewriteCond %{THE_REQUEST} \s/+index\.html[\s?] [NC]
RewriteRule ^index\.html$ https://gebaeudewerke.de/ [L,R=301]
RewriteCond %{THE_REQUEST} \s/+ratgeber/index\.php[\s?] [NC]
RewriteRule ^ratgeber/index\.php$ https://gebaeudewerke.de/ratgeber/ [L,R=301]

# 3. Seiten über auslieferung.php ausliefern: geplante Ratgeberbeiträge
#    erscheinen erst ab ihrem Datum, Links darauf ebenfalls.
RewriteRule ^$ /auslieferung.php [L]
RewriteCond %{REQUEST_URI} !^/404\.html$
RewriteCond %{REQUEST_FILENAME} -f
RewriteRule ^((?:ratgeber/)?[a-z0-9-]+\.html|llms\.txt)$ /auslieferung.php [L]
</IfModule>

# Komprimierung
<IfModule mod_deflate.c>
AddOutputFilterByType DEFLATE text/html text/css text/plain text/xml application/javascript text/javascript application/json application/ld+json application/xml image/svg+xml application/manifest+json image/x-icon
</IfModule>

# Zwischenspeichern im Browser
<IfModule mod_expires.c>
ExpiresActive On
ExpiresDefault "access plus 1 hour"
ExpiresByType text/html "access plus 0 seconds"
ExpiresByType text/css "access plus 1 month"
ExpiresByType application/javascript "access plus 1 month"
ExpiresByType text/javascript "access plus 1 month"
ExpiresByType image/png "access plus 6 months"
ExpiresByType image/webp "access plus 6 months"
ExpiresByType image/jpeg "access plus 1 year"
ExpiresByType image/svg+xml "access plus 6 months"
ExpiresByType image/x-icon "access plus 6 months"
ExpiresByType font/woff2 "access plus 1 year"
</IfModule>

<IfModule mod_headers.c>
Header always set X-Content-Type-Options "nosniff"
Header always set Referrer-Policy "strict-origin-when-cross-origin"
Header always set X-Frame-Options "SAMEORIGIN"
Header always set Permissions-Policy "geolocation=(), camera=(), microphone=()"
</IfModule>
